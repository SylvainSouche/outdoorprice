   132→   132→/** Normalise une couleur (supprime accents, souligné initial, pluriels simples). */
   133→   133→function canonColor(c: string): string {
   134→   134→  return c
   135→   135→    .toLowerCase()
   136→   136→    .trim()
   137→   137→    .replace(/^_+/, "")
   138→   138→    .normalize("NFD")
   139→   139→    .replace(/[\u0300-\u036f]/g, "")
   140→   140→    .replace(/s$/, ""); // marque → marques (best-effort, ne gère pas les exceptions)
   141→   141→}
   142→   142→
   143→   143→/** Trouve un hex pour un libellé couleur. Tente d'abord le libellé complet,
   144→   144→ *  puis chaque mot (ex: « fluo jaune » → jaune). */
   145→   145→function colorToHex(label: string): string | null {
   146→   146→  const canon = canonColor(label);
   147→   147→  if (COLOR_HEX[canon]) return COLOR_HEX[canon];
   148→   148→  // Tente chaque mot
   149→   149→  for (const part of canon.split(/[\s/-]+/)) {
   150→   150→    if (COLOR_HEX[part]) return COLOR_HEX[part];
   151→   151→  }
   152→   152→  return null;
   153→   153→}
