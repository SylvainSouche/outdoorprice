   538→interface LangContextValue {
   539→  lang: Lang;
   540→  setLang: (l: Lang) => void;
   541→  t: (key: string, params?: Record<string, string | number>) => string;
   542→}
   543→
   544→const LangContext = createContext<LangContextValue>({
   545→  lang: "fr",
   546→  setLang: () => {},
   547→  t: (key) => key,
   548→});
   549→
   550→export function LangProvider({ children }: { children: ReactNode }) {
   551→  // Lazy init from localStorage — avoids set-state-in-effect ESLint warning.
   552→  const [lang, setLangState] = useState<Lang>(() => {
   553→    if (typeof window === "undefined") return "fr";
   554→    const stored = localStorage.getItem(STORAGE_KEY) as Lang | null;
   555→    return stored && ["fr", "en", "es", "de"].includes(stored) ? stored : "fr";
   556→  });
   557→
   558→  const setLang = (l: Lang) => {
   559→    setLangState(l);
   560→    if (typeof window !== "undefined") {
   561→      localStorage.setItem(STORAGE_KEY, l);
   562→    }
   563→  };
   564→
   565→  const t = (key: string, params?: Record<string, string | number>): string => {
   566→    const dict = TRANSLATIONS[lang] ?? TRANSLATIONS.fr;
   567→    let str = dict[key] ?? TRANSLATIONS.fr[key] ?? key;
   568→    if (params) {
   569→      for (const [k, v] of Object.entries(params)) {
   570→        str = str.replace(new RegExp(`\\{${k}\\}`, "g"), String(v));
   571→      }
   572→    }
   573→    // Handle simple pluralization: {n} word{s} → "1 word" or "2 words"
   574→    // Pattern: {n} word{s} where {s} expands to "s" if n > 1, "" if n === 1
   575→    str = str.replace(/\{(\w+)\}(\w+)\{(\w+)\}/g, (_, countKey, singular, plural) => {
   576→      const count = params?.[countKey];
   577→      const n = typeof count === "number" ? count : 1;
   578→      return `${n} ${n !== 1 ? plural : singular}`;
   579→    });
   580→    return str;
   581→  };
   582→
   583→  return (
   584→    <LangContext.Provider value={{ lang, setLang, t }}>
   585→      {children}
   586→    </LangContext.Provider>
   587→  );
   588→}
   589→
   590→export function useLang() {
   591→  return useContext(LangContext);
   592→}
