   102→function formatPrice(p: number | null, currency: string): string {
   103→  if (p === null || p === undefined || Number.isNaN(p)) return "—";
   104→  if (currency === "EUR") {
   105→    return new Intl.NumberFormat("fr-FR", { style: "currency", currency: "EUR" }).format(p);
   106→  }
   107→  return `${currency} ${p.toFixed(2)}`;
   108→}

   132→/** Normalise une couleur (supprime accents, souligné initial, pluriels simples). */
   133→function canonColor(c: string): string {
   134→  return c
   135→    .toLowerCase()
   136→    .trim()
   137→    .replace(/^_+/, "")
   138→    .normalize("NFD")
   139→    .replace(/[\u0300-\u036f]/g, "")
   140→    .replace(/s$/, ""); // marque → marques (best-effort, ne gère pas les exceptions)
   141→}
   142→
   143→/** Trouve un hex pour un libellé couleur. Tente d'abord le libellé complet,
   144→ *  puis chaque mot (ex: « fluo jaune » → jaune). */
   145→function colorToHex(label: string): string | null {
   146→  const canon = canonColor(label);
   147→  if (COLOR_HEX[canon]) return COLOR_HEX[canon];
   148→  // Tente chaque mot
   149→  for (const part of canon.split(/[\s/-]+/)) {
   150→    if (COLOR_HEX[part]) return COLOR_HEX[part];
   151→  }
   152→  return null;
   153→}

   155→function getHostname(url: string): string {
   156→  try {
   157→    return new URL(url).hostname.replace(/^www\./, "");
   158→  } catch {
   159→    return url;
   160→  }

   979→function LogsPanel({ logs, onClear }: { logs: LogEntry[]; onClear: () => void }) {
   980→  const [expanded, setExpanded] = useState(false);
   981→  const [activeLevels, setActiveLevels] = useState<Set<LogLevel>>(new Set(["error", "warning", "info"]));
   982→  const [siteFilter, setSiteFilter] = useState<string>("");
   983→
   984→  const toggleLevel = (lvl: LogLevel) => {
   985→    setActiveLevels((prev) => {
   986→      const next = new Set(prev);
   987→      if (next.has(lvl)) next.delete(lvl);
   988→      else next.add(lvl);
   989→      return next;
   990→    });
   991→  };
   992→
   993→  const filtered = logs.filter((l) => {
   994→    if (!activeLevels.has(l.level)) return false;
   995→    if (siteFilter && !(l.site ?? "").toLowerCase().includes(siteFilter.toLowerCase())) return false;
   996→    return true;
   997→  });
   998→
   999→  const counts = {
  1000→    error: logs.filter((l) => l.level === "error").length,
  1001→    warning: logs.filter((l) => l.level === "warning").length,
  1002→    info: logs.filter((l) => l.level === "info").length,
  1003→  };
  1004→
  1005→  return (
  1006→    <div className={`fixed bottom-0 left-0 right-0 z-40 bg-white border-t-2 border-stone-300 shadow-2xl transition-transform duration-200 ${expanded ? "translate-y-0" : "translate-y-[calc(100%-2.5rem)]"}`}>
  1007→      {/* Header (always visible) */}

  1534→// ---------------------------------------------------------------------------
  1535→// GroupMatrixModal : popup avec vue matricielle (sites × groupes)
  1536→// --------------------------------------------------------------------------
  1537→// Chaque ligne = une boutique. Chaque colonne = un groupe (builtin ou custom).
  1538→// L'utilisateur coche/décoche les cases pour assigner une boutique à un ou
  1539→// plusieurs groupes. Les groupes builtin (outdoor, cycling, it) sont éditables
  1540→// aussi — ça permet par exemple d'ajouter Barrabes à "Cycling" si on veut.
  1541→// Le groupe "all" est exclu (il est implicite : tous les sites).
  1542→// --------------------------------------------------------------------------
  1543→
  1544→function GroupMatrixModal({
  1545→  groups,
  1546→  onGroupsChange,
  1547→  onClose,
  1548→}: {
  1549→  groups: ShopGroup[];
  1550→  onGroupsChange: (next: ShopGroup[]) => void;
  1551→  onClose: () => void;
  1552→}) {
  1553→  const [newGroupName, setNewGroupName] = useState("");
  1554→  const [newGroupIcon, setNewGroupIcon] = useState("");
  1555→
  1556→  // Groupes éditables = tous sauf "all" (qui est implicite)
  1557→  const editableGroups = groups.filter((g) => g.id !== "all");
  1558→
  1559→  // Helper : vérifie si un site est dans un groupe
  1560→  const isSiteInGroup = (siteId: SiteId, group: ShopGroup): boolean => {
  1561→    return group.sites.includes(siteId);
  1562→  };
  1563→
  1564→  // Toggle : ajoute ou retire un site d'un groupe
  1565→  const toggleSiteInGroup = (siteId: SiteId, groupId: string) => {
  1566→    const next = groups.map((g) => {
  1567→      if (g.id !== groupId) return g;
  1568→      const sites = new Set(g.sites);
  1569→      if (sites.has(siteId)) sites.delete(siteId);
  1570→      else sites.add(siteId);
  1571→      return { ...g, sites: Array.from(sites) };
  1572→    });
  1573→    onGroupsChange(next);
  1574→  };
  1575→
  1576→  // Crée un nouveau groupe custom
  1577→  const addCustomGroup = () => {
  1578→    const name = newGroupName.trim();
  1579→    if (!name) return;
  1580→    const id = `custom-${Date.now().toString(36)}`;
  1581→    const newGroup: ShopGroup = {
  1582→      id,
  1583→      name,
  1584→      icon: newGroupIcon.trim() || undefined,
  1585→      description: "Groupe personnalisé",
  1586→      accent: "bg-stone-100 text-stone-800 border-stone-300",
  1587→      sites: [],
  1588→      builtin: false,
  1589→    };
  1590→    onGroupsChange([...groups, newGroup]);
  1591→    setNewGroupName("");
  1592→    setNewGroupIcon("");
  1593→  };
  1594→
  1595→  // Supprime un groupe custom (uniquement les custom — pas les builtin)
  1596→  const removeGroup = (groupId: string) => {
  1597→    const group = groups.find((g) => g.id === groupId);
  1598→    if (!group || group.builtin) return;
  1599→    if (!confirm(`Supprimer le groupe « ${group.name} » ?`)) return;
  1600→    onGroupsChange(groups.filter((g) => g.id !== groupId));
  1601→  };
  1602→
  1603→  return (
  1604→    <div className="fixed inset-0 z-50 flex items-center justify-center bg-black/40 p-4" onClick={onClose}>
  1605→      <div
  1606→        className="bg-white rounded-lg shadow-2xl max-w-5xl w-full max-h-[90vh] flex flex-col"
  1607→        onClick={(e) => e.stopPropagation()}
  1608→      >
  1609→        {/* Header */}
  1610→        <div className="flex items-center gap-3 px-5 py-3 border-b border-stone-200">
  1611→          <Layers className="h-5 w-5 text-stone-700" />
  1612→          <h2 className="text-base font-semibold text-stone-900">Gérer les groupes</h2>
  1613→          <span className="text-xs text-stone-500">
  1614→            Vue matricielle : cochez les boutiques appartenant à chaque groupe
  1615→          </span>
  1616→          <button
  1617→            onClick={onClose}
  1618→            className="ml-auto text-stone-500 hover:text-stone-700"
  1619→            title="Fermer"
  1620→          >
  1621→            <X className="h-5 w-5" />
  1622→          </button>
  1623→        </div>
  1624→
  1625→        {/* Matrix */}
  1626→        <div className="overflow-auto flex-1 p-5">
  1627→          <table className="w-full text-sm border-collapse">
  1628→            <thead>
  1629→              <tr className="border-b-2 border-stone-300">
  1630→                <th className="text-left px-2 py-2 text-xs font-semibold text-stone-700 sticky left-0 bg-white">
  1631→                  Boutique
  1632→                </th>
  1633→                {editableGroups.map((g) => (
  1634→                  <th key={g.id} className="px-2 py-2 text-center min-w-[100px]">
  1635→                    <div className="flex flex-col items-center gap-0.5">
  1636→                      <span className="text-base">{g.icon ?? "•"}</span>
  1637→                      <span className={`text-[11px] font-medium px-1.5 py-0.5 rounded ${g.accent ?? ""}`}>
  1638→                        {g.name}
  1639→                      </span>
  1640→                      <span className="text-[10px] text-stone-500">
  1641→                        {g.sites.length} site{g.sites.length !== 1 ? "s" : ""}
  1642→                      </span>
  1643→                      {!g.builtin && (
  1644→                        <button
  1645→                          onClick={() => removeGroup(g.id)}
  1646→                          className="text-[10px] text-rose-500 hover:text-rose-700 mt-0.5"
  1647→                          title="Supprimer ce groupe"
  1648→                        >
  1649→                          supprimer
  1650→                        </button>
  1651→                      )}
  1652→                    </div>
  1653→                  </th>
  1654→                ))}

  1685→            </tbody>

  1687→        </div>
  1688→
  1689→        {/* Footer : créer un nouveau groupe */}
  1690→        <div className="border-t border-stone-200 px-5 py-3 bg-stone-50 flex items-center gap-2">

  1706→            maxLength={4}
  1707→          />
