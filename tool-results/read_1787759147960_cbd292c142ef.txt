   102→function formatPrice(p: number | null, currency: string): string {
   103→  if (p === null || p === undefined || Number.isNaN(p)) return "—";
   104→  if (currency === "EUR") {
   105→    return new Intl.NumberFormat("fr-FR", { style: "currency", currency: "EUR" }).format(p);
   106→  }
   107→  return `${currency} ${p.toFixed(2)}`;
   108→}

   132→/** Normalise une couleur (supprime accents, souligné initial, pluriels simples). */
   133→function canonColor(c: string): string {
   134→  return c
   135→    .toLowerCase()
   136→    .trim()
   137→    .replace(/^_+/, "")
   138→    .normalize("NFD")
   139→    .replace(/[\u0300-\u036f]/g, "")
   140→    .replace(/s$/, ""); // marque → marques (best-effort, ne gère pas les exceptions)
   141→}
   142→
   143→/** Trouve un hex pour un libellé couleur. Tente d'abord le libellé complet,
   144→ *  puis chaque mot (ex: « fluo jaune » → jaune). */
   145→function colorToHex(label: string): string | null {
   146→  const canon = canonColor(label);
   147→  if (COLOR_HEX[canon]) return COLOR_HEX[canon];
   148→  // Tente chaque mot
   149→  for (const part of canon.split(/[\s/-]+/)) {
   150→    if (COLOR_HEX[part]) return COLOR_HEX[part];
   151→  }
   152→  return null;
   153→}

   155→function getHostname(url: string): string {
   156→  try {
   157→    return new URL(url).hostname.replace(/^www\./, "");
   158→  } catch {
   159→    return url;
   160→  }

  1618→// aussi — ça permet par exemple d'ajouter Barrabes à "Cycling" si on veut.
  1619→// Le groupe "all" est exclu (il est implicite : tous les sites).

  1622→function GroupMatrixModal({
  1623→  groups,
  1624→  onGroupsChange,
  1625→  onClose,
  1626→}: {
  1627→  groups: ShopGroup[];
  1628→  onGroupsChange: (next: ShopGroup[]) => void;
  1629→  onClose: () => void;
  1630→}) {
  1631→  const [newGroupName, setNewGroupName] = useState("");
  1632→  const [newGroupIcon, setNewGroupIcon] = useState("");
  1633→
  1634→  // Groupes éditables = tous sauf "all" (qui est implicite)
  1635→  const editableGroups = groups.filter((g) => g.id !== "all");
  1636→
  1637→  // Helper : vérifie si un site est dans un groupe
  1638→  const isSiteInGroup = (siteId: SiteId, group: ShopGroup): boolean => {
  1639→    return group.sites.includes(siteId);
  1640→  };
  1641→
  1642→  // Toggle : ajoute ou retire un site d'un groupe
  1643→  const toggleSiteInGroup = (siteId: SiteId, groupId: string) => {
  1644→    const next = groups.map((g) => {
  1645→      if (g.id !== groupId) return g;
  1646→      const sites = new Set(g.sites);
  1647→      if (sites.has(siteId)) sites.delete(siteId);
  1648→      else sites.add(siteId);
  1649→      return { ...g, sites: Array.from(sites) };
  1650→    });
  1651→    onGroupsChange(next);
  1652→  };
  1653→
  1654→  // Crée un nouveau groupe custom
  1655→  const addCustomGroup = () => {
  1656→    const name = newGroupName.trim();
  1657→    if (!name) return;
  1658→    const id = `custom-${Date.now().toString(36)}`;
  1659→    const newGroup: ShopGroup = {
  1660→      id,
  1661→      name,
  1662→      icon: newGroupIcon.trim() || undefined,
  1663→      description: "Groupe personnalisé",
  1664→      accent: "bg-stone-100 text-stone-800 border-stone-300",
  1665→      sites: [],
  1666→      builtin: false,
  1667→    };
  1668→    onGroupsChange([...groups, newGroup]);
  1669→    setNewGroupName("");
  1670→    setNewGroupIcon("");
  1671→  };
  1672→
  1673→  // Supprime un groupe custom (uniquement les custom — pas les builtin)
  1674→  const removeGroup = (groupId: string) => {
  1675→    const group = groups.find((g) => g.id === groupId);
  1676→    if (!group || group.builtin) return;
  1677→    if (!confirm(`Supprimer le groupe « ${group.name} » ?`)) return;
  1678→    onGroupsChange(groups.filter((g) => g.id !== groupId));
  1679→  };
  1680→
  1681→  return (
  1682→    <div className="fixed inset-0 z-50 flex items-center justify-center bg-black/40 p-4" onClick={onClose}>
  1683→      <div
  1684→        className="bg-white rounded-lg shadow-2xl max-w-5xl w-full max-h-[90vh] flex flex-col"
  1685→        onClick={(e) => e.stopPropagation()}
  1686→      >
  1687→        {/* Header */}
  1688→        <div className="flex items-center gap-3 px-5 py-3 border-b border-stone-200">
  1689→          <Layers className="h-5 w-5 text-stone-700" />
  1690→          <h2 className="text-base font-semibold text-stone-900">Gérer les groupes</h2>
  1691→          <span className="text-xs text-stone-500">
  1692→            Vue matricielle : cochez les boutiques appartenant à chaque groupe
  1693→          </span>
  1694→          <button
  1695→            onClick={onClose}
  1696→            className="ml-auto text-stone-500 hover:text-stone-700"
  1697→            title="Fermer"
  1698→          >
  1699→            <X className="h-5 w-5" />
  1700→          </button>
  1701→        </div>
  1702→
  1703→        {/* Matrix */}
  1704→        <div className="overflow-auto flex-1 p-5">
  1705→          <table className="w-full text-sm border-collapse">
  1706→            <thead>
  1707→              <tr className="border-b-2 border-stone-300">
  1708→                <th className="text-left px-2 py-2 text-xs font-semibold text-stone-700 sticky left-0 bg-white">
  1709→                  Boutique
  1710→                </th>
  1711→                {editableGroups.map((g) => (
  1712→                  <th key={g.id} className="px-2 py-2 text-center min-w-[100px]">
  1713→                    <div className="flex flex-col items-center gap-0.5">
  1714→                      <span className="text-base">{g.icon ?? "•"}</span>
  1715→                      <span className={`text-[11px] font-medium px-1.5 py-0.5 rounded ${g.accent ?? ""}`}>
  1716→                        {g.name}
  1717→                      </span>
  1718→                      <span className="text-[10px] text-stone-500">
  1719→                        {g.sites.length} site{g.sites.length !== 1 ? "s" : ""}
  1720→                      </span>
